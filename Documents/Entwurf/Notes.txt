----- NoteModule

Note // Note entity with title and content
  -title: string //The title of the note
  -content: string //The content of the note
  -timeStamp: float //The time stamp (in seconds) of the lecture when the note was created. Used to jump back to the exact moment during playback
  +Note() //Default constructor for deserialization of Unity instantiation (Unity or JSON)
  +Note(title: string, content: string) //Constructor for general notes not linked any lecture.
  +Note(title: string, content: string, timestamp: float) //Constructor for notes linked to a specific time in the lecture.
  +GetTitle(): string //Gets the title of the note
  +GetContent(): string //Gets the content of the note
  +GetTimeStamp(): float //Gets the time stamp (in seconds) when the note was created.
  +SetTitle(title: string): void //Sets the title of the note
  +SetContent(content: string): void //Sets the content of the note
  +SetTimeStamp(time: float): void //Sets the time when the note was created (in seconds) during the lecture playback.

NoteManager // Manages note creation, editing, and storage
  -Notes: List<Note> //A list of all created notes and is used for viewing, editing, saving and deleting.
  +NoteManager() //Constructor of NoteManager
  +GetAllNotes(): List<Note> //Gets the list of created notes.
  +AddNote(note: Note): void //Adds the new note and stores the note to the list of notes.
  +DeleteNote(note: Note): void //Deletes the created note and deletes the note from the list of notes.
  +EditNote(note: Note, newTitle: string, newContent: string): void //Edits the given note by updating its title and/or content.
  +SaveNote(): void //Saves all notes locally in a JSON file for offline use.
  +LoadNote(): void //Loads saved notes from the local file for offline use.

NoteUtils // Utility class currently for time format conversion. Designed for future extensibility.
  -NoteUtils() //Private constructor to prevent instantiation of this utility class. 
  +ConvertSecondsToTimeString(seconds: float): string //Converts a time stamp in seconds to a readable time (in string) for GUI display (e.g., “00:01:25”). 
  +ConvertTimeStringToSeconds(time: string): float //Converts a readable time (in string) to seconds for internal use.

  -sceneId: string // The scene ID used to retrieve the localized name for display.
  -scenePath: string // The path to the corresponding Unity scene. (used for SceneManager.LoadScene() )

  <<Note>> ambientSound is optional (default: null)
  +GetSceneId (): string //Gets the scene ID of the VR environment.
